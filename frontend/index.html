/*This query retrieves student performance data including the average, lowest, highest, and median marks for each assessment in the Information Technology course at NP institution, segmented by branch and module. */ WITH MarksData AS ( SELECT i.InstitutionName, b.BranchName, c.CourseName, m.ModuleName, a.AssessId, a.AssessName, g.Marks, sub.SubmissionID FROM Institution i INNER JOIN Branch b ON i.InstituteId = b.InstitutionId INNER JOIN [User] u ON i.InstituteId = u.InstituteId INNER JOIN Student st ON u.UserId = st.StudentId INNER JOIN Submission sub ON u.UserId = sub.StudentId INNER JOIN Assessment a ON sub.AssessId = a.AssessId INNER JOIN Module m ON a.ModuleId = m.ModuleId INNER JOIN Course c ON m.CourseId = c.CourseId LEFT JOIN Grade g ON sub.GradeId = g.GradeId WHERE i.InstitutionName = 'NP' -- Filter by Institution (e.g., Ngee Ann Polytechnic) AND c.CourseName = 'Information Technology' -- Filter by Course name ), Aggregated AS ( SELECT Institution, Branch, Course, Module, AssessId AS AssessmentID, AssessName AS AssessmentName, COUNT(SubmissionID) AS SubmissionCount, AVG(Marks) AS AverageMarksOverall, AVG(CASE WHEN Marks IS NOT NULL THEN Marks END) AS AverageMarksCompleted, MIN(Marks) AS LowestMarks, MAX(Marks) AS HighestMarks, PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY Marks) OVER (PARTITION BY Institution, Branch, Course, Module, AssessId, AssessName) AS MedianMarks FROM MarksData GROUP BY Institution, Branch, Course, Module, AssessId, AssessName ) SELECT Institution, Branch, Course, Module, AssessmentID, AssessmentName, SubmissionCount, AverageMarksOverall, AverageMarksCompleted, LowestMarks, HighestMarks, MedianMarks FROM Aggregated;
